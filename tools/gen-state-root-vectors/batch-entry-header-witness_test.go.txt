	"github.com/zenon-network/go-zenon/zenon/mock"
)

func TestSPVBatchEntryHeaderWitnessCapture(t *testing.T) {
	saveSporkState(t)
	z := mock.NewMockZenon(t)
	t.Cleanup(z.StopPanic)
	activateOffchainExecution(t, z)
	saveBridgeConstants(t)
	oldAdmin := types.NetworkAdmin
	types.NetworkAdmin = g.User5.Address
	t.Cleanup(func() { types.NetworkAdmin = oldAdmin })

	// The original fixture helpers restore these globals on cleanup. This is
	// an attestation-profile fixture, not an STF execution or delay experiment.
	constants.MinSoftDelay = 10
	constants.MinAdministratorDelay = 10
	oldWithdrawalDelay := constants.WithdrawalDelayMin
	constants.WithdrawalDelayMin = 10
	t.Cleanup(func() { constants.WithdrawalDelayMin = oldWithdrawalDelay })

	activateStateRoot(t, z)
	// Check the selected Settlement operations only. The original convenience
	// helper starts this capture before the extra state-root setup and would
	// fail at cleanup on its legitimate spork-activation diagnostic messages.
	logs := z.SaveLogs(common.EmbeddedLogger)
	t.Cleanup(func() { logs.Equals(t, "") })
	registerDomains(t, z, 1)
	settlementCall(t, z, setApprovedDAModesStep(g.User5.Address, 1, []byte{0}), nil)
	advanceMomentums(t, z, constants.MinSoftDelay)
	settlementCall(t, z, setApprovedDAModesStep(g.User5.Address, 1, []byte{0}), nil)
	configureAsset(t, z, 1, assetIdFor(types.ZnnTokenStandard))
	registerExecutorFully(t, z, g.Pillar1Name, g.User1.Address)

	// Use unchanged contract fixtures to store one actual ABI row. Only public
	// deterministic test keypairs are used in this in-process test chain.
	settlementCall(t, z, depositStep(g.User1.Address, types.ZnnTokenStandard, 7, 1, types.Hash{0x07}, 5000), nil)
	post := hashFromByteForTest(0x51)
	settlementCall(t, z, submitBatchStep(g.User1.Address,
		batchCommitment(1, 0, 0, 0, types.ZeroHash, post, g.User1.Address, znnDepositSummary(7))), nil)
	entry, err := definition.GetBatchEntry(settlementStorage(t, z), 1, 0)
	if err != nil || entry == nil || entry.PostStateRoot != post || entry.ExecutorPillar != g.Pillar1Name {
		t.Fatal("selected fixture BatchEntry was not stored", err)
	}

	ledger := api.NewLedgerApi(z)
	h, err := ledger.GetFrontierMomentum()
	if err != nil || h == nil || h.Version != nom.StateRootMomentumVersion || h.StateRoot.IsZero() || h.Height < 4 || h.Height > 256 {
		t.Fatal("bounded fixture v3 header unavailable", err)
	}
	key := common.JoinBytes([]byte{0x03}, types.SettlementContract.Bytes(), []byte{0x04}, entry.Key())
	if len(key) != 35 {
		t.Fatal("full ledger key width changed")
	}
	path := types.NewHash(key)
	witness, err := ledger.GetProof(h.Height, key)
	if err != nil || witness == nil || witness.Root != h.StateRoot || len(witness.Value) == 0 || len(witness.Value) > 4096 || len(witness.Proof) > 12391 {
		t.Fatal("bounded original-node witness unavailable", err)
	}
	ok, err := trie.VerifyProofByPath(h.StateRoot, trie.Path(path), witness.Value, witness.Proof)
	if err != nil || !ok {
		t.Fatal("original node proof did not match its signed header", err)
	}
	decoded := new(definition.BatchEntry)
	if err := definition.ABISettlement.UnpackVariable(decoded, definition.BatchEntryVariableName, witness.Value); err != nil || decoded.PostStateRoot != post {
		t.Fatal("original ABI decoder did not match the stored row", err)
	}
	tampered := append([]byte(nil), witness.Value...)
	tampered[len(tampered)-1] ^= 1
	if ok, _ := trie.VerifyProofByPath(h.StateRoot, trie.Path(path), tampered, witness.Proof); ok {
		t.Fatal("altered value retained proof membership")
	}

	absentKey := common.JoinBytes([]byte{0x03}, types.SettlementContract.Bytes(), []byte{0x04},
		(&definition.BatchEntryKey{DomainId: 1, BatchId: 1}).Key())
	absent, err := ledger.GetProof(h.Height, absentKey)
	if err != nil || absent == nil || absent.Root != h.StateRoot || absent.Value != nil {
		t.Fatal("separate missing batch fixture changed", err)
	}
	if ok, err := trie.VerifyAbsence(h.StateRoot, absentKey, absent.Proof); err != nil || !ok {
		t.Fatal("original absence control failed", err)
	}

	z.InsertNewMomentum()
	tip, err := ledger.GetFrontierMomentum()
	if err != nil || tip.Height != h.Height+1 || tip.PreviousHash != h.Hash {
		t.Fatal("fixture tip linkage changed", err)
	}
	repeated, err := ledger.GetProof(h.Height, key)
	if err != nil || repeated.Root != witness.Root || !bytes.Equal(repeated.Value, witness.Value) || !bytes.Equal(repeated.Proof, witness.Proof) {
		t.Fatal("historical witness changed after one extension", err)
	}
	rows, err := ledger.GetMomentumsByHeight(h.Height-1, 3)
	if err != nil || rows == nil || len(rows.List) != 3 {
		t.Fatal("actual RPC momentum projection unavailable", err)
	}
	for _, row := range rows.List {
		if row.Version != 3 || verifier.MomentumStatic(row.Momentum) != nil {
			t.Fatal("captured header hash or fixture signature failed")
		}
	}
	store := z.Chain().GetFrontierMomentumStore()
	anchor, err := store.GetMomentumByHeight(h.Height - 2)
	if err != nil || anchor == nil || rows.List[0].PreviousHash != anchor.Hash {
		t.Fatal("fixture trust-start identity unavailable", err)
	}
	genesis, err := store.GetMomentumByHeight(1)
	if err != nil || genesis == nil || genesis.Version != 1 {
		t.Fatal("fixture genesis unavailable", err)
	}
	var firstV3 uint64
	for height := uint64(2); height <= h.Height; height++ {
		m, err := store.GetMomentumByHeight(height)
		if err != nil || m == nil {
			t.Fatal("fixture boundary observation unavailable", err)
		}
		if m.Version == 3 {
			firstV3 = height
			break
		}
	}
	if firstV3 == 0 || firstV3 > anchor.Height {
		t.Fatal("fixture boundary not observed before selected anchor")
	}
	gateway, err := embedded.NewSettlementApi(z).GetBatch(1, 0)
	if err != nil || gateway == nil || gateway.PostStateRoot != post {
		t.Fatal("original gateway DTO unavailable", err)
	}
	report := map[string]interface{}{
		"format":   "selected-node-batch-entry-header-witness/v1",
		"scope":    "ISOLATED_ATTESTATION_FIXTURE_NOT_SELECTED_NETWORK_OR_STF_EXECUTION",
		"chain_id": z.Chain().ChainIdentifier(), "genesis_hash": genesis.Hash.String(),
		"first_incoming_v3": firstV3,
		"anchor":            map[string]interface{}{"height": anchor.Height, "hash": anchor.Hash.String()},
		"height":            h.Height, "header_hash": h.Hash.String(), "tip_height": tip.Height, "tip_hash": tip.Hash.String(),
		"domain_id": uint32(1), "batch_id": uint64(0), "ledger_key": hex.EncodeToString(key), "path": path.String(),
		"bundle":          map[string]interface{}{"version": 1, "momentums": rows.List, "witness": witness, "gateway": gateway},
		"absent_batch_id": uint64(1), "absent_witness": absent,
		"original_proof_and_ABI_checked": true, "tampered_value_refused": true, "historical_witness_stable": true,
		"fixture_public_keys_only": true, "RPC_P2P_service_started": false, "network_execution": false,
		"selected_network_profile_or_anchor": false, "STF_execution_or_finality_established": false,
	}
	raw, err := json.Marshal(report)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("SPV_BATCH_ENTRY_HEADER_WITNESS %s", raw)
}
