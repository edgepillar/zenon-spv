type spvCountedDB struct {
	db.DB
	gets, puts, putValueBytes uint64
}

func (d *spvCountedDB) Get(key []byte) ([]byte, error) {
	d.gets++
	return d.DB.Get(key)
}

func (d *spvCountedDB) Put(key, value []byte) error {
	d.puts++
	d.putValueBytes += uint64(len(value))
	return d.DB.Put(key, value)
}

func spvPointCount(t *testing.T, d db.DB, prefix byte) uint64 {
	t.Helper()
	it := d.NewIterator([]byte{prefix})
	defer it.Release()
	var count uint64
	for it.Next() {
		count++
	}
	if it.Error() != nil {
		t.Fatal("consensus point iteration failed")
	}
	return count
}

type spvFirstPeriodReadGate struct {
	db.DB
	corechain.MomentumEventListener
	pointKey         []byte
	entered, release chan struct{}
	notification     types.HashHeight
	seconds          float64
}

func (g *spvFirstPeriodReadGate) Get(key []byte) ([]byte, error) {
	if g.pointKey != nil && bytes.Equal(key, g.pointKey) {
		g.pointKey = nil
		close(g.entered)
		<-g.release
	}
	return g.DB.Get(key)
}

func (g *spvFirstPeriodReadGate) InsertMomentum(momentum *nom.DetailedMomentum) {
	g.notification = momentum.Momentum.Identifier()
	started := time.Now()
	g.MomentumEventListener.InsertMomentum(momentum)
	g.seconds = time.Since(started).Seconds()
}

// Read the Ledger API's frontier store while the original points listener is
// at its first period database read. The channel holds that exact read.
func spvObserveInsertionVisibility(t *testing.T, ch corechain.Chain, first, next types.HashHeight,
	gate *spvFirstPeriodReadGate, insert func() error) {
	t.Helper()
	before, err := ch.GetFrontierMomentumStore().GetFrontierMomentum()
	if err != nil || before == nil || before.Identifier() != first {
		t.Fatal("unexpected frontier before the unsigned insertion")
	}
	gate.pointKey = storage.CreatePointKey(storage.PrefixPeriodPoint, 0)
	gate.entered = make(chan struct{})
	gate.release = make(chan struct{})
	ch.Register(gate)
	timer := time.NewTimer(8 * time.Second)
	defer timer.Stop()
	insertDone := make(chan struct{})
	var insertError error
	go func() {
		insertError = insert()
		close(insertDone)
	}()
	var readDone chan struct{}
	released := false
	defer func() {
		if !released {
			close(gate.release)
		}
		<-insertDone
		if readDone != nil {
			<-readDone
		}
		ch.UnRegister(gate)
	}()

	select {
	case <-gate.entered:
		if gate.notification != next {
			t.Fatal("unexpected fixture insertion notification")
		}
	case <-insertDone:
		t.Fatal("unsigned insertion returned before the first period read")
	case <-timer.C:
		t.Fatal("first period database read deadline")
	}

	readDone = make(chan struct{})
	var during *nom.Momentum
	var readError error
	go func() {
		during, readError = ch.GetFrontierMomentumStore().GetFrontierMomentum()
		close(readDone)
	}()
	select {
	case <-readDone:
		if readError != nil || during == nil || during.Identifier() != next {
			t.Fatal("unsigned fixture frontier is not visible during the callback")
		}
	case <-timer.C:
		t.Fatal("frontier read blocked behind the insertion callback")
	}
	select {
	case <-insertDone:
		t.Fatal("unsigned insertion returned while the callback was held")
	default:
	}

	close(gate.release)
	released = true
	select {
	case <-insertDone:
		if insertError != nil {
			t.Fatal("unsigned fixture insertion failed after the callback")
		}
	case <-timer.C:
		t.Fatal("unsigned insertion did not return after the callback")
	}
	row := map[string]interface{}{
		"format":        "first-insertion-frontier-visibility/v2",
		"before_height": before.Height, "during_height": during.Height,
		"callback_held_during_read": true, "insert_returned_after_release": true,
		"first_period_read_held":    true,
		"accepted_header_authority": false,
	}
	data, err := json.Marshal(row)
	if err != nil {
		t.Fatal("frontier visibility observation encoding failed")
	}
	t.Logf("SPV_FIRST_INSERTION_VISIBILITY %s", data)
}

func TestSPVFirstInsertionCatchup(t *testing.T) {
	// Do not start a node wrapper, consensus event loop, producer, RPC or peer.
	// In particular, do not construct a wallet or obtain a signing key.
	log15.Root().SetHandler(log15.DiscardHandler())
	raw, err := os.ReadFile(filepath.Join("..", "docker", "devnet", "genesis.json"))
	if err != nil {
		t.Fatal("selected genesis unavailable")
	}
	sum := sha256.Sum256(raw)
	const sourceSHA = "0dd33b18b147f70b3d3d5f07199a2b45311f6e9b84a98b3f86d8f6c6b45f419a"
	if len(raw) != 8137 || hex.EncodeToString(sum[:]) != sourceSHA {
		t.Fatal("selected public genesis changed")
	}
	var config genesis.GenesisConfig
	if json.Unmarshal(raw, &config) != nil || genesis.CheckGenesis(&config) != nil || config.ChainIdentifier != 69 {
		t.Fatal("selected public genesis invalid")
	}
	if constants.ConsensusConfig.BlockTime != 10 || constants.ConsensusConfig.NodeCount != 30 {
		t.Fatal("selected consensus period changed")
	}
	for _, selected := range []struct {
		name string
		gap  uint64
	}{
		{"one-period", 300},
		{"one-day", 86400},
		// 2026-11-15T10:30:00Z minus the unchanged source genesis timestamp.
		{"fixture-window", 54815400},
	} {
		t.Run(selected.name, func(t *testing.T) {
			root := t.TempDir()
			ch := corechain.NewChain(db.NewLevelDBManager(filepath.Join(root, "ledger")),
				cache.NewCacheDBManager(filepath.Join(root, "cache")), genesis.NewGenesis(&config),
				filepath.Join(root, "state"), false)
			stopped := false
			defer func() {
				if !stopped && ch.Stop() != nil {
					t.Error("owned chain cleanup failed")
				}
			}()
			if ch.Init() != nil {
				t.Fatal("owned chain initialization failed")
			}
			first := ch.GetGenesisMomentum()
			if first.Height != 1 || first.TimestampUnix != 1739923200 || first.Hash.String() != "fab7609bae7fd78eb7d2b0d4a5023832f1a6730921b862f2415da25524698dc9" {
				t.Fatal("selected genesis identity differs")
			}
			// The manager accepts an explicit test fixture without authenticating it.
			// This is deliberately below the verifier, with no signature and no network.
			patch := db.NewPatch()
			m := &nom.Momentum{Version: 1, ChainIdentifier: 69, Height: 2,
				PreviousHash: first.Hash, TimestampUnix: first.TimestampUnix + selected.gap,
				ChangesHash: db.PatchHash(patch)}
			m.Hash = m.ComputeHash()
			m.EnsureCache()
			detailed := &nom.DetailedMomentum{Momentum: m}
			t.Log("SPV_CATCHUP_PHASE begin unsigned-manager-fixture")
			insertFixture := func() error {
				insert := ch.AcquireInsert("isolated unsigned catchup cost fixture")
				defer insert.Unlock()
				if err := ch.UpdateCache(insert, detailed, patch); err != nil {
					return err
				}
				if err := ch.UpdateStateTree(insert, detailed, patch); err != nil {
					return err
				}
				return ch.AddMomentumTransaction(insert, &nom.MomentumTransaction{Momentum: m, Changes: patch})
			}
			var insertSeconds float64
			if selected.name != "one-period" {
				started := time.Now()
				if insertFixture() != nil {
					t.Fatal("unsigned manager fixture insertion failed")
				}
				insertSeconds = time.Since(started).Seconds()
				if ch.GetFrontierMomentumStore().Identifier() != m.Identifier() {
					t.Fatal("unsigned fixture frontier differs")
				}
			}
			rawDB, ldb := db.NewLevelDB(filepath.Join(root, "consensus"))
			closed := false
			defer func() {
				if !closed && ldb.Close() != nil {
					t.Error("owned consensus cleanup failed")
				}
			}()
			counted := &spvCountedDB{DB: rawDB}
			var pointDB db.DB = counted
			var gate *spvFirstPeriodReadGate
			if selected.name == "one-period" {
				gate = &spvFirstPeriodReadGate{DB: counted}
				pointDB = gate
			}
			listener, completed := consensus.SPVFirstInsertionPoints(pointDB, ch)
			previousPeriod, previousEpoch := completed()
			if previousPeriod != -1 || previousEpoch != -1 {
				t.Fatal("fresh point database required")
			}
			counted.gets, counted.puts, counted.putValueBytes = 0, 0, 0
			profile := os.Getenv(spvCPUProfileEnv)
			observeCPU := profile != "" && selected.name == "fixture-window"
			var cpuBefore spvCPUCounters
			if observeCPU {
				cpuBefore = spvCPUStats(t, profile)
			}
			t.Log("SPV_CATCHUP_PHASE begin original-points-listener")
			started := time.Now()
			var pointSeconds float64
			if gate != nil {
				gate.MomentumEventListener = listener
				spvObserveInsertionVisibility(t, ch, first.Identifier(), m.Identifier(), gate, insertFixture)
				insertSeconds = time.Since(started).Seconds()
				pointSeconds = gate.seconds
			} else {
				listener.InsertMomentum(detailed)
				pointSeconds = time.Since(started).Seconds()
			}
			t.Log("SPV_CATCHUP_PHASE end original-points-listener")
			if observeCPU {
				spvCPUReport(t, profile, cpuBefore, spvCPUStats(t, profile))
			}
			periods, epochs := selected.gap/300, selected.gap/86400
			periodCount := spvPointCount(t, rawDB, storage.PrefixPeriodPoint)
			epochCount := spvPointCount(t, rawDB, storage.PrefixEpochPoint)
			completedPeriod, completedEpoch := completed()
			if periodCount != periods || epochCount != epochs || completedPeriod != int64(periods)-1 || completedEpoch != int64(epochs)-1 {
				t.Fatal("original listener did not complete the selected history")
			}
			if ldb.Close() != nil {
				t.Fatal("consensus database close failed")
			}
			closed = true
			if ch.Stop() != nil {
				t.Fatal("chain manager close failed")
			}
			stopped = true
			var files, logicalBytes uint64
			if filepath.WalkDir(filepath.Join(root, "consensus"), func(_ string, entry fs.DirEntry, err error) error {
				if err != nil {
					return err
				}
				if entry.IsDir() {
					return nil
				}
				info, err := entry.Info()
				if err != nil {
					return err
				}
				if !info.Mode().IsRegular() {
					t.Fatal("unexpected nonregular consensus fixture")
				}
				files++
				logicalBytes += uint64(info.Size())
				return nil
			}) != nil {
				t.Fatal("closed consensus file measurement failed")
			}
			report := map[string]interface{}{
				"format": "selected-first-insertion-catchup/v1", "case": selected.name,
				"node_revision": "2835c436b197fece5c76adf4ba8dc1decca2e3ea", "source_genesis_sha256": sourceSHA,
				"source_genesis_unix": first.TimestampUnix, "unsigned_H2_unix": m.TimestampUnix,
				"gap_seconds": selected.gap, "period_seconds": 300, "completed_periods": periodCount, "completed_epochs": epochCount,
				"unsigned_insert_seconds": insertSeconds, "points_listener_seconds": pointSeconds,
				"consensus_get_calls": counted.gets, "consensus_put_calls": counted.puts, "consensus_put_value_bytes": counted.putValueBytes,
				"closed_consensus_files": files, "closed_consensus_logical_bytes": logicalBytes,
				"original_points_listener_completed": true, "real_LevelDB_managers": true, "owned_databases_closed": true,
				"unsigned_test_fixture_bypasses_authentication": true, "Chain_Start": false, "consensus_Start": false,
				"wallet_or_signing": false, "network_or_broadcast": false, "actual_fixture_cause_established": false,
				"accepted_header_authority": false, "new_devnet_acceptance_established": false, "peak_disk_or_RSS_measured": false,
			}
			encoded, err := json.Marshal(report)
			if err != nil {
				t.Fatal("observation encoding failed")
			}
			t.Logf("SPV_FIRST_INSERTION_CATCHUP %s", encoded)
		})
	}
}
