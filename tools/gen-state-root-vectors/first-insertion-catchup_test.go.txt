type spvCountedDB struct {
	db.DB
	gets, puts, putValueBytes uint64
}

func (d *spvCountedDB) Get(key []byte) ([]byte, error) {
	d.gets++
	return d.DB.Get(key)
}

func (d *spvCountedDB) Put(key, value []byte) error {
	d.puts++
	d.putValueBytes += uint64(len(value))
	return d.DB.Put(key, value)
}

func spvPointCount(t *testing.T, d db.DB, prefix byte) uint64 {
	t.Helper()
	it := d.NewIterator([]byte{prefix})
	defer it.Release()
	var count uint64
	for it.Next() {
		count++
	}
	if it.Error() != nil {
		t.Fatal("consensus point iteration failed")
	}
	return count
}

type spvFirstPeriodReadGate struct {
	db.DB
	corechain.MomentumEventListener
	pointKey         []byte
	entered, release chan struct{}
	notification     types.HashHeight
	seconds          float64
}

func (g *spvFirstPeriodReadGate) Get(key []byte) ([]byte, error) {
	if g.pointKey != nil && bytes.Equal(key, g.pointKey) {
		g.pointKey = nil
		close(g.entered)
		<-g.release
	}
	return g.DB.Get(key)
}

func (g *spvFirstPeriodReadGate) InsertMomentum(momentum *nom.DetailedMomentum) {
	g.notification = momentum.Momentum.Identifier()
	started := time.Now()
	g.MomentumEventListener.InsertMomentum(momentum)
	g.seconds = time.Since(started).Seconds()
}

func spvCheckRecordedStartupHeader(t *testing.T, ch corechain.Chain, oracle consensus.Consensus, genesis types.HashHeight) *nom.Momentum {
	t.Helper()
	raw := []byte(spvRecordedStartupHeader)
	pin := sha256.Sum256(raw)
	if len(raw) != 697 || hex.EncodeToString(pin[:]) != "931a991acef939bd4d9c0f55e293c64402189940b406b612a0e8bed9668acbc4" {
		t.Fatal("recorded startup frontier response changed")
	}
	var reply struct {
		Version string        `json:"jsonrpc"`
		ID      uint64        `json:"id"`
		Result  *api.Momentum `json:"result"`
	}
	if err := json.Unmarshal(raw, &reply); err != nil || reply.Version != "2.0" ||
		reply.ID != 32 || reply.Result == nil || reply.Result.Momentum == nil {
		t.Fatal("invalid recorded startup frontier response")
	}
	header := reply.Result.Momentum
	if header.Version != 1 || header.ChainIdentifier != 69 || header.Height != 2 ||
		header.Previous() != genesis || header.ComputeHash() != header.Hash || !header.StateRoot.IsZero() {
		t.Fatal("recorded header does not match the source genesis and signed envelope")
	}
	if len(header.PublicKey) != ed25519.PublicKeySize ||
		!ed25519.Verify(ed25519.PublicKey(header.PublicKey), header.Hash.Bytes(), header.Signature) ||
		reply.Result.Producer != header.Producer() {
		t.Fatal("recorded header signature or producer address differs")
	}
	header.EnsureCache()
	matches, err := oracle.VerifyMomentumProducerAt(genesis, header)
	if err != nil || !matches {
		t.Fatal("recorded header producer does not match the source genesis slot")
	}
	if ch.GetFrontierMomentumStore().Identifier() != genesis {
		t.Fatal("source frontier changed while checking the recorded header")
	}
	trace := map[string]interface{}{
		"format": "recorded-chain69-h2/v1", "recorded_run": 43653383092, "recorded_artifact_id": 14136634774,
		"rpc_sha256": hex.EncodeToString(pin[:]), "height": header.Height, "version": header.Version,
		"hash": header.Hash.String(), "parent_hash": genesis.Hash.String(), "timestamp": header.TimestampUnix,
		"signature_verified": true, "source_parent_slot_match": matches, "source_frontier_unchanged": true,
		"accepted_header_authority": false,
	}
	data, err := json.Marshal(trace)
	if err != nil {
		t.Fatal("recorded startup header observation encoding failed")
	}
	t.Logf("SPV_RECORDED_STARTUP_H2 %s", data)
	return header
}

	return corechain.NewChain(db.NewLevelDBManager(filepath.Join(root, "ledger")),

		filepath.Join(root, "state"), false)
}

func spvInsertStartupMomentum(ch corechain.Chain, detailed *nom.DetailedMomentum, changes db.Patch) error {
	insert := ch.AcquireInsert("isolated startup momentum fixture")
	defer insert.Unlock()
	if err := ch.UpdateCache(insert, detailed, changes); err != nil {
		return err
	}
	if err := ch.UpdateStateTree(insert, detailed, changes); err != nil {
		return err
	}
	return ch.AddMomentumTransaction(insert, &nom.MomentumTransaction{Momentum: detailed.Momentum, Changes: changes})
}

type spvStartupWitness struct {
	Key             string `json:"key"`
	Proof           []byte `json:"proof"`
	Value           []byte `json:"value"`
	ProofRPCRefused bool   `json:"proof_rpc_refused"`
}

type spvStartupProofPin struct {
	SHA256 string `json:"sha256"`
	Bytes  int    `json:"bytes"`
}

func spvSettlementKeys() map[string][]byte {
	const domainID = 1
	subkeys := map[string][]byte{
		"batch_entry":   (&definition.BatchEntryKey{DomainId: domainID, BatchId: 0}).Key(),
		"domain_record": (&definition.DomainRecordKey{DomainId: domainID}).Key(),
		// The selected source keeps the domainConfig key constructor private.
		"domain_config": binary.BigEndian.AppendUint32([]byte{0x15}, domainID),
	}
	for name, subkey := range subkeys {
		subkeys[name] = common.JoinBytes([]byte{0x03}, types.SettlementContract.Bytes(), []byte{0x04}, subkey)
	}
	return subkeys
}

func spvReplayedSettlementProofs(t *testing.T, ch corechain.Chain, header *nom.Momentum, root types.Hash) ([]byte, map[string]spvStartupWitness) {
	t.Helper()
	ledger := api.NewLedgerApi(&spvFrontierNode{chain: ch})
	rpcRoot, err := ledger.GetStateRoot(header.Height)
	if !errors.Is(err, api.ErrStateRootNotActivated) || !rpcRoot.IsZero() {
		t.Fatal("legacy H2 exposed a root through the ledger API")
	}

	keys := spvSettlementKeys()
	raw := make(map[string]spvStartupWitness, len(keys))
	for _, name := range []string{"batch_entry", "domain_record", "domain_config"} {
		key := keys[name]
		proofResult, err := ledger.GetProof(header.Height, key)
		if !errors.Is(err, api.ErrStateRootNotActivated) || proofResult != nil {
			t.Fatalf("legacy H2 supplied a proof for %s through the ledger API", name)
		}
		value, proof, err := ch.GetProof(header.Identifier(), key)
		if err != nil || value != nil || len(proof) < 99 || len(proof) > 99+256*32 {
			t.Fatalf("bounded local absence proof is unavailable for %s", name)
		}
		if matches, err := trie.VerifyAbsence(root, key, proof); err != nil || !matches {
			t.Fatalf("local tree proof did not match for %s", name)
		}
		if matches, _ := trie.VerifyAbsence(header.StateRoot, key, proof); matches {
			t.Fatalf("local proof unexpectedly matched the legacy header for %s", name)
		}
		raw[name] = spvStartupWitness{Key: hex.EncodeToString(key), Proof: proof, Value: value, ProofRPCRefused: true}
	}
	trace := map[string]interface{}{
		"format": "recorded-chain69-h2-settlement-proofs/v1", "height": header.Height, "hash": header.Hash.String(),
		"local_tree_root": root.String(), "header_state_root": header.StateRoot.String(),
		"root_rpc_refused": true, "witnesses": raw, "accepted_header_authority": false,
	}
	encoded, err := json.Marshal(trace)
	if err != nil {
		t.Fatal("cannot encode the local Settlement proofs")
	}
	return encoded, raw
}

func spvObserveReopenedH3(t *testing.T, ch corechain.Chain, candidate *nom.Momentum, prior map[string]spvStartupWitness) {
	t.Helper()
	frontier, err := ch.GetFrontierMomentumStore().GetFrontierMomentum()
	if err != nil || frontier == nil || frontier.Identifier() != candidate.Identifier() ||
		frontier.ComputeHash() != candidate.Hash || frontier.StateRoot != candidate.StateRoot ||
		len(frontier.PublicKey) != 0 || len(frontier.Signature) != 0 || !ch.StateTreeReady() {
		t.Fatal("cannot reopen the isolated unsigned H3 fixture")
	}
	ledger := api.NewLedgerApi(&spvFrontierNode{chain: ch})
	root, err := ledger.GetStateRoot(candidate.Height)
	if err != nil || root != candidate.StateRoot {
		t.Fatal("H3 Ledger root differs from the reopened header root")
	}
	if legacy, err := ledger.GetStateRoot(candidate.Height - 1); !errors.Is(err, api.ErrStateRootNotActivated) || !legacy.IsZero() {
		t.Fatal("the later v3 fixture exposed a root at H2")
	}
	pins := spvCheckHistoricalAbsence(t, ledger, candidate, prior)
	trace := map[string]interface{}{
		"format": "chain69-unsigned-h3-absence-rpc/v1", "height": candidate.Height, "version": candidate.Version,
		"hash": candidate.Hash.String(), "parent_hash": candidate.PreviousHash.String(), "timestamp": candidate.TimestampUnix,
		"header_state_root": candidate.StateRoot.String(), "ledger_root": root.String(), "proof_pins": pins,
		"reopened_fixture": true, "unsigned_vm_refused": true, "h2_remains_refused": true,
		"accepted_header_authority": false,
	}
	encoded, err := json.Marshal(trace)
	if err != nil {
		t.Fatal("cannot encode the H3 Ledger observation")
	}
	t.Logf("SPV_UNSIGNED_H3_LEDGER %s", encoded)
}

func spvCheckHistoricalAbsence(t *testing.T, ledger *api.LedgerApi, target *nom.Momentum, prior map[string]spvStartupWitness) map[string]spvStartupProofPin {
	t.Helper()
	pins := make(map[string]spvStartupProofPin, len(prior))
	for _, name := range []string{"batch_entry", "domain_record", "domain_config"} {
		original, present := prior[name]
		key, err := hex.DecodeString(original.Key)
		if err != nil || !present {
			t.Fatalf("missing selected H2 key for %s", name)
		}
		if legacy, err := ledger.GetProof(target.Height-1, key); !errors.Is(err, api.ErrStateRootNotActivated) || legacy != nil {
			t.Fatalf("the later v3 fixture exposed a proof at H2 for %s", name)
		}
		witness, err := ledger.GetProof(target.Height, key)
		if err != nil || witness == nil || witness.Value != nil || witness.Root != target.StateRoot ||
			!bytes.Equal(witness.Proof, original.Proof) {
			t.Fatalf("H3 Ledger absence witness differs for %s", name)
		}
		if matches, err := trie.VerifyAbsence(target.StateRoot, key, witness.Proof); err != nil || !matches {
			t.Fatalf("H3 Ledger witness failed against the header root for %s", name)
		}
		sum := sha256.Sum256(witness.Proof)
		pins[name] = spvStartupProofPin{SHA256: hex.EncodeToString(sum[:]), Bytes: len(witness.Proof)}
	}
	return pins
}

func spvCheckH3AtH4(t *testing.T, ch corechain.Chain, target, tip *nom.Momentum, markerKey, marker []byte,
	prior map[string]spvStartupWitness) map[string]spvStartupProofPin {
	t.Helper()
	ledger := api.NewLedgerApi(&spvFrontierNode{chain: ch})
	frontier, err := ledger.GetFrontierMomentum()
	if err != nil || frontier == nil || frontier.Identifier() != tip.Identifier() ||
		frontier.PreviousHash != target.Hash || frontier.StateRoot != tip.StateRoot ||
		len(frontier.PublicKey) != 0 || len(frontier.Signature) != 0 || !ch.StateTreeReady() {
		t.Fatal("isolated H4 frontier or state tree differs")
	}
	historical, err := ledger.GetStateRoot(target.Height)
	if err != nil || historical != target.StateRoot || historical == tip.StateRoot {
		t.Fatal("historical Ledger root differs from the H3 header")
	}
	current, err := ledger.GetStateRoot(tip.Height)
	if err != nil || current != tip.StateRoot {
		t.Fatal("H4 Ledger root differs from the fixture tip")
	}
	historicalMarker, err := ledger.GetProof(target.Height, markerKey)
	if err != nil || historicalMarker == nil || historicalMarker.Root != historical || historicalMarker.Value != nil {
		t.Fatal("tip-only fixture marker appeared in the historical proof")
	}
	if matches, err := trie.VerifyAbsence(historical, markerKey, historicalMarker.Proof); err != nil || !matches {
		t.Fatal("historical marker absence proof differs")
	}
	currentMarker, err := ledger.GetProof(tip.Height, markerKey)
	if err != nil || currentMarker == nil || currentMarker.Root != current || !bytes.Equal(currentMarker.Value, marker) {
		t.Fatal("fixture marker is unavailable at the H4 tip")
	}
	if matches, err := trie.VerifyProofByPath(current, trie.Path(types.NewHash(markerKey)), marker, currentMarker.Proof); err != nil || !matches {
		t.Fatal("tip marker membership proof differs")
	}
	return spvCheckHistoricalAbsence(t, ledger, target, prior)
}

type spvTypedHistoryRow struct {
	key   []byte
	value []byte
}

func spvTypedHistoryRows(t *testing.T) map[string]spvTypedHistoryRow {
	t.Helper()
	raw, err := os.ReadFile("spv_batch_context_node_witness.json")
	digest := sha256.Sum256(raw)
	if err != nil || hex.EncodeToString(digest[:]) != "773cda402e08709ffd73adfea355f73269b1470b7336c5a027ddd5b331ab9d96" {
		t.Fatal("typed row source changed", err)
	}
	var fixture struct {
		Rows []struct {
			Name  string `json:"name"`
			Key   string `json:"ledger_key_hex"`
			Value string `json:"value_hex"`
		}
	}
	if json.Unmarshal(raw, &fixture) != nil || len(fixture.Rows) != 4 || fixture.Rows[3].Name != "absent_batch" {
		t.Fatal("typed row source encoding differs")
	}
	keys := spvSettlementKeys()
	rows := make(map[string]spvTypedHistoryRow, len(keys))
	for _, rawRow := range fixture.Rows[:3] {
		key, exists := keys[rawRow.Name]
		value, err := hex.DecodeString(rawRow.Value)
		if _, duplicate := rows[rawRow.Name]; !exists || duplicate || rawRow.Key != hex.EncodeToString(key) ||
			err != nil || len(value) == 0 || len(value) > 4096 {
			t.Fatal("unexpected typed row source", rawRow.Name)
		}
		rows[rawRow.Name] = spvTypedHistoryRow{key: key, value: value}
	}
	return rows
}

func spvAppendUnsignedHistory(t *testing.T, ch corechain.Chain, parent *nom.Momentum, changes db.Patch) *nom.Momentum {
	t.Helper()
	root, err := ch.ComputeStateRoot(parent.Identifier(), changes)
	if err != nil || root.IsZero() || root == parent.StateRoot {
		t.Fatal("history fixture must have a distinct state root")
	}
	header := &nom.Momentum{
		Version: nom.StateRootMomentumVersion, ChainIdentifier: parent.ChainIdentifier,
		Height: parent.Height + 1, PreviousHash: parent.Hash,
		TimestampUnix: parent.TimestampUnix + uint64(constants.ConsensusConfig.BlockTime),
		StateRoot:     root, ChangesHash: db.PatchHash(changes),
		NextFusionPrice: dp.MinResourcePrice, NextWorkPrice: dp.MinResourcePrice,
	}
	header.EnsureCache()
	header.Hash = header.ComputeHash()
	if spvInsertStartupMomentum(ch, &nom.DetailedMomentum{Momentum: header}, changes) != nil {
		t.Fatal("cannot extend the isolated historical-read fixture")
	}
	return header
}

func spvReadTypedHistory(t *testing.T, ledger *api.LedgerApi, target *nom.Momentum,
	rows map[string]spvTypedHistoryRow) map[string][]byte {
	t.Helper()
	root, err := ledger.GetStateRoot(target.Height)
	if err != nil || root != target.StateRoot {
		t.Fatal("typed historical root differs from the selected header")
	}
	proofs := make(map[string][]byte, len(rows))
	for name, row := range rows {
		witness, err := ledger.GetProof(target.Height, row.key)
		if err != nil || witness == nil || witness.Root != root || !bytes.Equal(witness.Value, row.value) ||
			len(witness.Proof) < 103 || len(witness.Proof) > 12391 {
			t.Fatalf("bounded typed membership is unavailable for %s", name)
		}
		if matches, err := trie.VerifyProofByPath(root, trie.Path(types.NewHash(row.key)), row.value, witness.Proof); err != nil || !matches {
			t.Fatalf("typed membership does not match the selected root for %s", name)
		}
		proofs[name] = witness.Proof
	}
	return proofs
}

func spvPruneClosedFixtureTree(t *testing.T, stateDir string, target, tip *nom.Momentum) {
	t.Helper()
	ldb, err := leveldb.OpenFile(stateDir, &opt.Options{ErrorIfMissing: true})
	if err != nil {
		t.Fatal("cannot open the closed fixture tree", err)
	}
	defer func() {
		if err := ldb.Close(); err != nil {
			t.Error("cannot close the pruned fixture tree", err)
		}
	}()
	tree, err := trie.NewNodeTree(ldb)
	if err != nil || tree.FrontierIdentifier() != tip.Identifier() || target.Height+1 != tip.Height {
		t.Fatal("unexpected fixture tree before pruning", err)
	}
	if root, err := tree.Root(target.Identifier()); err != nil || root != target.StateRoot {
		t.Fatal("typed history must be retained before pruning", err)
	}
	if err := tree.Prune(tip.Height); err != nil {
		t.Fatal("cannot prune the closed fixture below its tip", err)
	}
	if root, err := tree.Root(target.Identifier()); !errors.Is(err, trie.ErrNoVersion) || !root.IsZero() {
		t.Fatal("pruned typed history still has a tree version", err)
	}
	if root, err := tree.Root(tip.Identifier()); err != nil || root != tip.StateRoot {
		t.Fatal("retained tip changed during pruning", err)
	}
}

func spvStateTreeRPC(t *testing.T, rpc *server.Server, method string, params ...interface{}) []byte {
	t.Helper()
	body, err := json.Marshal(map[string]interface{}{
		"jsonrpc": "2.0", "id": 1, "method": method, "params": params,
	})
	if err != nil {
		t.Fatal("cannot encode fixture RPC request", err)
	}
	request := httptest.NewRequest(http.MethodPost, "/", bytes.NewReader(body))
	request.Header.Set("Content-Type", "application/json")
	reply := httptest.NewRecorder()
	rpc.ServeHTTP(reply, request)
	if reply.Code != http.StatusOK || reply.Header().Get("Content-Type") != "application/json" ||
		reply.Body.Len() == 0 || reply.Body.Len() > 16<<10 {
		t.Fatal("unexpected state tree HTTP reply")
	}
	return reply.Body.Bytes()
}

func spvObservePrunedTypedHistory(t *testing.T, ch corechain.Chain, target, tip *nom.Momentum, absence []byte) {
	t.Helper()
	store := ch.GetFrontierMomentumStore()
	old, err := store.GetMomentumByHeight(target.Height)
	if err != nil || old == nil || old.Identifier() != target.Identifier() || old.StateRoot != target.StateRoot ||
		store.Identifier() != tip.Identifier() || !ch.StateTreeReady() {
		t.Fatal("reopened fixture lost its ledger header or retained frontier", err)
	}
	rpc := server.NewServer()
	defer rpc.Stop()
	if err := rpc.RegisterName("ledger", api.NewLedgerApi(&spvFrontierNode{chain: ch})); err != nil {
		t.Fatal("cannot register the original ledger API", err)
	}
	pruned := spvStateTreeRPC(t, rpc, "ledger.getStateRoot", target.Height)
	const prunedError = `{"jsonrpc":"2.0","id":1,"error":{"code":-32000,"message":"trie: no such version"}}` + "\n"
	if string(pruned) != prunedError {
		t.Fatal("pruned root RPC did not return the expected version error")
	}
	keys := spvSettlementKeys()
	checked := map[string]string{}
	for _, name := range []string{"batch_entry", "domain_record", "domain_config"} {
		key := keys[name]
		proof := spvStateTreeRPC(t, rpc, "ledger.getProof", target.Height, key)
		if !bytes.Equal(proof, pruned) {
			t.Fatalf("pruned witness RPC did not return the version error for %s", name)
		}
		checked[name] = hex.EncodeToString(key)
	}
	retained := spvStateTreeRPC(t, rpc, "ledger.getProof", tip.Height, keys["batch_entry"])
	var response struct {
		Version string          `json:"jsonrpc"`
		ID      uint64          `json:"id"`
		Error   json.RawMessage `json:"error"`
		Result  *api.StateProof `json:"result"`
	}
	if json.Unmarshal(retained, &response) != nil || response.Version != "2.0" || response.ID != 1 ||
		response.Error != nil || response.Result == nil || response.Result.Root != tip.StateRoot ||
		response.Result.Value != nil || !bytes.Equal(response.Result.Proof, absence) {
		t.Fatal("retained tip RPC did not preserve the original batch absence proof")

// The production sender and decoder share this in-memory writer. The signed
// envelope comes from the archived RPC reply and no peer service is started.
type spvStartupMessage struct {
	detailed *nom.DetailedMomentum
	size     uint32
	digest   string
}

func (r *spvStartupMessage) WriteMsg(message p2p.Msg) error {
	if message.Code != protocol.NewBlockMsg || message.Size == 0 || message.Size > 4096 {
		return errors.New("recorded startup message is outside its bounds")
	}
	source := &io.LimitedReader{R: message.Payload, N: int64(message.Size)}
	digest := sha256.New()
	message.Payload = io.TeeReader(source, digest)
	if err := message.Decode(&r.detailed); err != nil {
		return err
	}
	if source.N != 0 || r.detailed == nil || r.detailed.Momentum == nil {
		return errors.New("recorded startup message is incomplete")
	}
	r.detailed.Momentum.EnsureCache()
	r.size, r.digest = message.Size, hex.EncodeToString(digest.Sum(nil))
	return nil
}

const spvFrontierMethod = "ledger.getFrontierMomentum"

type spvFetchNotification struct {
	entered chan types.HashHeight
	release <-chan struct{}
}

func (g *spvFetchNotification) InsertMomentum(detailed *nom.DetailedMomentum) {
	g.entered <- detailed.Momentum.Identifier()
	<-g.release
}

func (*spvFetchNotification) DeleteMomentum(*nom.DetailedMomentum) {}

// Enqueue the recorded block in the original fetcher. The final callback
// represents the handoff to the broadcaster, with no transport behind it.
func spvObserveRecordedFetch(t *testing.T, ch corechain.Chain, bridge protocol.ChainBridge,
	detailed *nom.DetailedMomentum, first types.HashHeight) {
	t.Helper()
	next := detailed.Momentum.Identifier()
	rpc := server.NewServer()
	defer rpc.Stop()
	if err := rpc.RegisterName("ledger", &spvFrontierLedger{ledger: api.NewLedgerApi(&spvFrontierNode{chain: ch})}); err != nil {
		t.Fatal("cannot register the read-only ledger method", err)
	}
	before, err := spvReadRPCFrontier(rpc)
	if err != nil || before != first || bridge.GetBlock(next.Hash) != nil {
		t.Fatal("unexpected initial frontier or known recorded block", err)
	}

	deadline, cancel := context.WithTimeout(context.Background(), 8*time.Second)
	gate := &spvFetchNotification{entered: make(chan types.HashHeight, 1), release: deadline.Done()}
	ch.Register(gate)
	type relay struct {
		id        types.HashHeight
		propagate bool
	}
	ready, settled := make(chan struct{}), make(chan struct{})
	finished, relayed := make(chan error, 1), make(chan relay, 1)
	startupReads := 0
	lookupStarted := false
	queue := fetcher.New(func(hash types.Hash) *nom.DetailedMomentum {
		if hash == next.Hash {
			// The loop looks up H2. The worker looks up its H1 parent.
			lookupStarted = true
		}
		return bridge.GetBlock(hash)
	}, bridge.VerifyMomentum, func(block *nom.DetailedMomentum, propagate bool) {
		relayed <- relay{block.Momentum.Identifier(), propagate}
	}, func() uint64 {
		height := bridge.CurrentBlock().Height
		// The pinned loop reads height at startup and after each of its
		// two initial expired timers. Once drained, the sole event after
		// it takes H2 off the queue is the import worker's done signal.
		startupReads++
		if startupReads == 3 {
			close(ready)
		}
		if lookupStarted {
			close(settled)
			lookupStarted = false
		}
		return height
	}, func(blocks []*nom.DetailedMomentum) (int, error) {
		index, err := bridge.InsertChain(blocks)
		if err == nil && index != 0 {
			err = errors.New("unexpected recorded H2 insertion index")
		}
		finished <- err
		return index, err
	}, func(string) {})
	queue.Start()
	submitted := false
	defer func() {
		cancel()
		// Let the original loop consume the worker's completion before
		// stopping it and closing the fixture chain.
		if submitted {
			<-settled
		} else {
			<-ready
		}
		queue.Stop()
		ch.UnRegister(gate)
	}()
	select {
	case <-ready:
	case <-deadline.Done():
		t.Fatal("recorded fetch queue startup deadline")
	}
	if err := queue.Enqueue("recorded-peer", detailed); err != nil {
		t.Fatal("cannot enqueue recorded H2", err)
	}
	submitted = true
	select {
	case id := <-gate.entered:
		if id != next {
			t.Fatal("unexpected recorded fetch notification", id)
		}
	case err := <-finished:
		t.Fatal("recorded fetch insertion returned before its notification", err)
	case <-settled:
		t.Fatal("recorded fetch settled without notifying insertion")
	case <-deadline.Done():
		t.Fatal("recorded fetch notification deadline")
	}
	during, err := spvReadRPCFrontier(rpc)
	if err != nil || during != next || deadline.Err() != nil {
		t.Fatal("RPC did not expose recorded H2 while its notification was held", err)
	}
	select {
	case err := <-finished:
		t.Fatal("recorded fetch insertion returned while its notification was held", err)
	case <-relayed:
		t.Fatal("recorded fetch called the broadcaster while its notification was held")
	default:
	}
	cancel()

	completion := time.NewTimer(8 * time.Second)
	defer completion.Stop()
	select {
	case err := <-finished:
		if err != nil || ch.GetFrontierMomentumStore().Identifier() != next {
			t.Fatal("recorded fetch did not commit H2", err)
		}
	case <-completion.C:
		t.Fatal("recorded fetch insertion completion deadline")
	}
	var broadcast relay
	select {
	case broadcast = <-relayed:
		if broadcast.id != next || !broadcast.propagate {
			t.Fatal("recorded fetch sent an unexpected block to the broadcaster")
		}
	case <-completion.C:
		t.Fatal("recorded fetch broadcast callback deadline")
	}

	data, err := json.Marshal(map[string]interface{}{
		"format": "recorded-chain69-peer-fetch/v1", "rpc_method": spvFrontierMethod,
		"before_height": before.Height, "before_hash": before.Hash.String(),
		"held_height": during.Height, "held_hash": during.Hash.String(),
		"insert_returned_while_held": false, "relay_called_while_held": false,
		"relayed_hash": broadcast.id.Hash.String(), "relay_propagate": broadcast.propagate,
		"network_started": false, "accepted_header_authority": false,
	})
	if err != nil {
		t.Fatal("cannot encode the recorded fetch observation", err)
	}
	t.Logf("SPV_RECORDED_PEER_FETCH %s", data)
}

func spvReadRPCFrontier(rpc *server.Server) (types.HashHeight, error) {
	body := `{"jsonrpc":"2.0","id":1,"method":"` + spvFrontierMethod + `","params":[]}`
	request := httptest.NewRequest(http.MethodPost, "/", strings.NewReader(body))
	request.Header.Set("Content-Type", "application/json")
	reply := httptest.NewRecorder()
	rpc.ServeHTTP(reply, request)
	if reply.Code != http.StatusOK || reply.Header().Get("Content-Type") != "application/json" ||
		reply.Body.Len() == 0 || reply.Body.Len() > 16<<10 {
		return types.HashHeight{}, errors.New("unexpected frontier HTTP reply")
	}
	var response struct {
		Version string            `json:"jsonrpc"`
		ID      uint64            `json:"id"`
		Result  *types.HashHeight `json:"result"`
		Error   json.RawMessage   `json:"error"`
	}
	if err := json.Unmarshal(reply.Body.Bytes(), &response); err != nil || response.Version != "2.0" ||
		response.ID != 1 || response.Result == nil || response.Error != nil {
		return types.HashHeight{}, errors.New("unexpected frontier JSON-RPC reply")
	}
	return *response.Result, nil
}

// Read the actual readiness RPC while the original points listener is at its
// first period database read. The HTTP handler runs entirely in memory.
func spvObserveInsertionVisibility(t *testing.T, ch corechain.Chain, first, next types.HashHeight,
	gate *spvFirstPeriodReadGate, insert func() error) {
	t.Helper()
	rpc := server.NewServer()
	defer rpc.Stop()
	if err := rpc.RegisterName("ledger", &spvFrontierLedger{ledger: api.NewLedgerApi(&spvFrontierNode{chain: ch})}); err != nil {
		t.Fatal("cannot register the read-only ledger method")
	}
	before, err := spvReadRPCFrontier(rpc)
	if err != nil || before != first {
		t.Fatal("unexpected RPC frontier before the unsigned insertion")
	}
	gate.pointKey = storage.CreatePointKey(storage.PrefixPeriodPoint, 0)
	gate.entered = make(chan struct{})
	gate.release = make(chan struct{})
	ch.Register(gate)
	timer := time.NewTimer(8 * time.Second)
	defer timer.Stop()
	insertDone := make(chan struct{})
	var insertError error
	go func() {
		insertError = insert()
		close(insertDone)
	}()
	var readDone chan struct{}
	released := false
	defer func() {
		if !released {
			close(gate.release)
		}
		<-insertDone
		if readDone != nil {
			<-readDone
		}
		ch.UnRegister(gate)
	}()

	select {
	case <-gate.entered:
		if gate.notification != next {
			t.Fatal("unexpected fixture insertion notification")
		}
	case <-insertDone:
		t.Fatal("unsigned insertion returned before the first period read")
	case <-timer.C:
		t.Fatal("first period database read deadline")
	}

	readDone = make(chan struct{})
	var during types.HashHeight
	var readError error
	go func() {
		during, readError = spvReadRPCFrontier(rpc)
		close(readDone)
	}()
	select {
	case <-readDone:
		if readError != nil || during != next {
			t.Fatal("unsigned fixture frontier is not visible through RPC during the callback")
		}
	case <-timer.C:
		t.Fatal("RPC frontier read blocked behind the insertion callback")
	}
	select {
	case <-insertDone:
		t.Fatal("unsigned insertion returned while the callback was held")
	default:
	}

	close(gate.release)
	released = true
	select {
	case <-insertDone:
		if insertError != nil {
			t.Fatal("unsigned fixture insertion failed after the callback")
		}
	case <-timer.C:
		t.Fatal("unsigned insertion did not return after the callback")
	}
	row := map[string]interface{}{
		"format":        "first-insertion-frontier-visibility/v3",
		"before_height": before.Height, "during_height": during.Height,
		"callback_held_during_read": true, "insert_returned_after_release": true,
		"first_period_read_held": true, "rpc_method": spvFrontierMethod, "rpc_transport": "in-memory-http",
		"accepted_header_authority": false,
	}
	data, err := json.Marshal(row)
	if err != nil {
		t.Fatal("frontier visibility observation encoding failed")
	}
	t.Logf("SPV_FIRST_INSERTION_VISIBILITY %s", data)
}

	// Do not start a node wrapper, consensus event loop, producer or peer.
	// In particular, do not construct a wallet or obtain a signing key.
	log15.Root().SetHandler(log15.DiscardHandler())
	raw, err := os.ReadFile(filepath.Join("..", "docker", "devnet", "genesis.json"))
	if err != nil {
		t.Fatal("selected genesis unavailable")
	}
	sum := sha256.Sum256(raw)
	const sourceSHA = "0dd33b18b147f70b3d3d5f07199a2b45311f6e9b84a98b3f86d8f6c6b45f419a"
	if len(raw) != 8137 || hex.EncodeToString(sum[:]) != sourceSHA {
		t.Fatal("selected public genesis changed")
	}
	var config genesis.GenesisConfig
	if json.Unmarshal(raw, &config) != nil || genesis.CheckGenesis(&config) != nil || config.ChainIdentifier != 69 {
		t.Fatal("selected public genesis invalid")
	}
	if constants.ConsensusConfig.BlockTime != 10 || constants.ConsensusConfig.NodeCount != 30 {
		t.Fatal("selected consensus period changed")
	}
	for _, selected := range []struct {
		name string
		gap  uint64
	}{
		{"one-period", 300},
		{"one-day", 86400},
		// 2026-11-15T10:30:00Z minus the unchanged source genesis timestamp.
		{"fixture-window", 54815400},
	} {
		t.Run(selected.name, func(t *testing.T) {
			root := t.TempDir()
			ch := spvNewStartupChain(root, &config)
			stopped := false
			defer func() {
				if !stopped && ch.Stop() != nil {
					t.Error("owned chain cleanup failed")
				}
			}()
			if ch.Init() != nil {
				t.Fatal("owned chain initialization failed")
			}
			first := ch.GetGenesisMomentum()
			if first.Height != 1 || first.TimestampUnix != 1739923200 || first.Hash.String() != "fab7609bae7fd78eb7d2b0d4a5023832f1a6730921b862f2415da25524698dc9" {
				t.Fatal("selected genesis identity differs")
			}
			if selected.name == "one-period" {
				spvReplayRecordedStartupBoundary(t, &config, first.Identifier())
			}
			// The manager accepts an explicit test fixture without authenticating it.
			// This is deliberately below the verifier, with no signature and no network.
			patch := db.NewPatch()
			m := &nom.Momentum{Version: 1, ChainIdentifier: 69, Height: 2,
				PreviousHash: first.Hash, TimestampUnix: first.TimestampUnix + selected.gap,
				ChangesHash: db.PatchHash(patch)}
			if selected.name == "one-period" {
				// The RPC converter fills nil optional slices. Seal the empty fields
				// before the insertion listener and the read can see this fixture.
				m.Data = []byte{}
				m.Content = nom.MomentumContent{}
			}
			m.Hash = m.ComputeHash()
			m.EnsureCache()
			detailed := &nom.DetailedMomentum{Momentum: m}
			t.Log("SPV_CATCHUP_PHASE begin unsigned-manager-fixture")
			insertFixture := func() error {
				return spvInsertStartupMomentum(ch, detailed, patch)
			}
			var insertSeconds float64
			if selected.name != "one-period" {
				started := time.Now()
				if insertFixture() != nil {
					t.Fatal("unsigned manager fixture insertion failed")
				}
				insertSeconds = time.Since(started).Seconds()
				if ch.GetFrontierMomentumStore().Identifier() != m.Identifier() {
					t.Fatal("unsigned fixture frontier differs")
				}
			}
			rawDB, ldb := db.NewLevelDB(filepath.Join(root, "consensus"))
			closed := false
			defer func() {
				if !closed && ldb.Close() != nil {
					t.Error("owned consensus cleanup failed")
				}
			}()
			counted := &spvCountedDB{DB: rawDB}
			var pointDB db.DB = counted
			var gate *spvFirstPeriodReadGate
			if selected.name == "one-period" {
				gate = &spvFirstPeriodReadGate{DB: counted}
				pointDB = gate
			}
			listener, completed := consensus.SPVFirstInsertionPoints(pointDB, ch)
			previousPeriod, previousEpoch := completed()
			if previousPeriod != -1 || previousEpoch != -1 {
				t.Fatal("fresh point database required")
			}
			counted.gets, counted.puts, counted.putValueBytes = 0, 0, 0
			profile := os.Getenv(spvCPUProfileEnv)
			observeCPU := profile != "" && selected.name == "fixture-window"
			var cpuBefore spvCPUCounters
			if observeCPU {
				cpuBefore = spvCPUStats(t, profile)
			}
			t.Log("SPV_CATCHUP_PHASE begin original-points-listener")
			started := time.Now()
			var pointSeconds float64
			if gate != nil {
				gate.MomentumEventListener = listener
				spvObserveInsertionVisibility(t, ch, first.Identifier(), m.Identifier(), gate, insertFixture)
				insertSeconds = time.Since(started).Seconds()
				pointSeconds = gate.seconds
			} else {
				listener.InsertMomentum(detailed)
				pointSeconds = time.Since(started).Seconds()
			}
			t.Log("SPV_CATCHUP_PHASE end original-points-listener")
			if observeCPU {
				spvCPUReport(t, profile, cpuBefore, spvCPUStats(t, profile))
			}
			periods, epochs := selected.gap/300, selected.gap/86400
			periodCount := spvPointCount(t, rawDB, storage.PrefixPeriodPoint)
			epochCount := spvPointCount(t, rawDB, storage.PrefixEpochPoint)
			completedPeriod, completedEpoch := completed()
			if periodCount != periods || epochCount != epochs || completedPeriod != int64(periods)-1 || completedEpoch != int64(epochs)-1 {
				t.Fatal("original listener did not complete the selected history")
			}
			if ldb.Close() != nil {
				t.Fatal("consensus database close failed")
			}
			closed = true
			if ch.Stop() != nil {
				t.Fatal("chain manager close failed")
			}
			stopped = true
			var files, logicalBytes uint64
			if filepath.WalkDir(filepath.Join(root, "consensus"), func(_ string, entry fs.DirEntry, err error) error {
				if err != nil {
					return err
				}
				if entry.IsDir() {
					return nil
				}
				info, err := entry.Info()
				if err != nil {
					return err
				}
				if !info.Mode().IsRegular() {
					t.Fatal("unexpected nonregular consensus fixture")
				}
				files++
				logicalBytes += uint64(info.Size())
				return nil
			}) != nil {
				t.Fatal("closed consensus file measurement failed")
			}
			report := map[string]interface{}{
				"format": "selected-first-insertion-catchup/v1", "case": selected.name,
				"node_revision": "2835c436b197fece5c76adf4ba8dc1decca2e3ea", "source_genesis_sha256": sourceSHA,
				"source_genesis_unix": first.TimestampUnix, "unsigned_H2_unix": m.TimestampUnix,
				"gap_seconds": selected.gap, "period_seconds": 300, "completed_periods": periodCount, "completed_epochs": epochCount,
				"unsigned_insert_seconds": insertSeconds, "points_listener_seconds": pointSeconds,
				"consensus_get_calls": counted.gets, "consensus_put_calls": counted.puts, "consensus_put_value_bytes": counted.putValueBytes,
				"closed_consensus_files": files, "closed_consensus_logical_bytes": logicalBytes,
				"original_points_listener_completed": true, "real_LevelDB_managers": true, "owned_databases_closed": true,
				"unsigned_test_fixture_bypasses_authentication": true, "Chain_Start": false, "consensus_Start": false,
				"wallet_or_signing": false, "network_or_broadcast": false, "actual_fixture_cause_established": false,
				"accepted_header_authority": false, "new_devnet_acceptance_established": false, "peak_disk_or_RSS_measured": false,
			}
			encoded, err := json.Marshal(report)
			if err != nil {
				t.Fatal("observation encoding failed")
			}
			t.Logf("SPV_FIRST_INSERTION_CATCHUP %s", encoded)
		})
	}
}
