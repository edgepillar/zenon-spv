// Read the limits and counters of the transient group running this test.
package consensus_test

import (
	"encoding/json"
	"os"
	"path/filepath"
	"runtime"
	"strconv"
	"strings"
	"testing"
)

const spvCPUProfileEnv = "SPV_CATCHUP_CPU_PROFILE"

type spvCPUCounters struct {
	UsageUsec     uint64 `json:"usage_usec"`
	Periods       uint64 `json:"nr_periods"`
	Throttled     uint64 `json:"nr_throttled"`
	ThrottledUsec uint64 `json:"throttled_usec"`
}

func spvCPUStats(t *testing.T, profile string) spvCPUCounters {
	t.Helper()
	read := func(path string) string {
		raw, err := os.ReadFile(path)
		if err != nil || len(raw) > 4096 {
			t.Fatal("kernel resource observation unavailable")
		}
		return strings.TrimSpace(string(raw))
	}
	if runtime.GOOS != "linux" || runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("expected Linux and one Go scheduler worker")
	}
	group, ok := strings.CutPrefix(read("/proc/self/cgroup"), "0::")
	unit := os.Getenv("SPV_CATCHUP_CPU_UNIT")
	if !ok || unit == "" || filepath.Dir(group) != "/system.slice" || filepath.Base(group) != unit {
		t.Fatal("test is outside its selected cgroup v2 unit")
	}
	limits := map[string]string{"reference": "max 100000", "half-core": "50000 100000"}
	limit, ok := limits[profile]
	if !ok {
		t.Fatal("unexpected CPU observation profile")
	}
	dir := filepath.Join("/sys/fs/cgroup", strings.TrimPrefix(group, "/"))
	if read(filepath.Join(dir, "cpu.max")) != limit ||
		read(filepath.Join(dir, "memory.max")) != "805306368" ||
		read(filepath.Join(dir, "memory.swap.max")) != "0" {
		t.Fatal("selected CPU or memory limit not observed")
	}
	values := make(map[string]uint64)
	for _, line := range strings.Split(read(filepath.Join(dir, "cpu.stat")), "\n") {
		fields := strings.Fields(line)
		if len(fields) != 2 {
			t.Fatal("invalid CPU counters")
		}
		value, err := strconv.ParseUint(fields[1], 10, 64)
		_, duplicate := values[fields[0]]
		if err != nil || duplicate {
			t.Fatal("invalid CPU counter value")
		}
		values[fields[0]] = value
	}
	get := func(key string) uint64 {
		value, ok := values[key]
		if !ok {
			t.Fatal("missing CPU counter", key)
		}
		return value
	}
	return spvCPUCounters{get("usage_usec"), get("nr_periods"), get("nr_throttled"), get("throttled_usec")}
}

func spvCPUReport(t *testing.T, profile string, before, after spvCPUCounters) {
	t.Helper()
	if after.UsageUsec < before.UsageUsec || after.Periods < before.Periods ||
		after.Throttled < before.Throttled || after.ThrottledUsec < before.ThrottledUsec {
		t.Fatal("CPU counters decreased during listener")
	}
	cpuMax := "max 100000"
	if profile == "half-core" {
		cpuMax = "50000 100000"
	}
	raw, err := json.Marshal(map[string]any{
		"format":  "first-insertion-cpu-observation/v1",
		"profile": profile, "cpu_max": cpuMax, "memory_max_bytes": 805306368,
		"memory_swap_max_bytes": 0, "gomaxprocs": runtime.GOMAXPROCS(0),
		"listener_cpu_delta": spvCPUCounters{
			after.UsageUsec - before.UsageUsec, after.Periods - before.Periods,
			after.Throttled - before.Throttled, after.ThrottledUsec - before.ThrottledUsec,
		},
	})
	if err != nil {
		t.Fatal("CPU observation encoding failed")
	}
	t.Logf("SPV_FIRST_INSERTION_CPU %s", raw)
}
